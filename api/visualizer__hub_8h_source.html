138 VisualizerHub(const std::string& system_name,
139 const std::optional<VisualizerOptions>& options = std::nullopt)
140 : system_name_ {system_name},
141 data_path_ {data_path_root_ / system_name_} {
142 options_ = std::make_unique<VisualizerOptions>();
143 if (options) {
144 logging::log()->info("VH:Constructor: Using passed visualizer options");
145 options_ = std::make_unique<VisualizerOptions>(*options);
146 } else {
147 logging::log()->warn(
148 "VH:Constructor: No visualizer_options.yaml found; using "
149 "defaults.");
150 }
151 // Check for default init
152 if (options_->default_init) {
153 logging::log()->info("VH:Constructor: Default init enabled");
154 std::string deployment;
155 const char* result = std::getenv("DEPLOYMENT");
156 if (result == nullptr || result[0] == '\0') {
157 throw std::runtime_error(
158 "VH:Constructor: DEPLOYMENT environment variable not set for "
159 "default init!");
160 }
161 deployment = common::utils::to_lower(std::string(result));
162 logging::log()->info(
163 "VH:Constructor: Loading default context for deployment {}",
164 deployment);
165 const auto context_it = options_->default_contexts.find(deployment);
166 if (context_it == options_->default_contexts.end()) {
167 throw std::runtime_error(
168 "VH:Constructor: No default context found for deployment "
169 + deployment);
170 }
171 VisualizerData visualizer_data;
172 visualizer_data.context_id = draco::PlanContextId {context_it->second};
173
174 if (!LoadAndSetModelData(visualizer_data)) {
175 throw std::runtime_error(
176 "VH:Constructor: Failed to load default model on initialization!");
177 }
178 }
179 }

233 bool HasButton(const std::string_view button_name) const {

236 return std::find(button_names.begin(), button_names.end(),

241 void ResetButtons() {
242 DRAKE_THROW_UNLESS(draco_ != nullptr);
243 for (const auto& name : draco_->robot_model().meshcat()->GetButtonNames()) {
244 draco_->robot_model().meshcat()->DeleteButton(name);
245 }
246 for (const auto& [name, keycode] : button_order_) {
247 draco_->robot_model().meshcat()->AddButton(name, keycode);
248 }
249 }
250
251 void DeleteButton(const std::string_view button_name) {
252 DRAKE_THROW_UNLESS(draco_ != nullptr);
253 // Remove from button order
254 button_order_.erase(
255 std::remove_if(button_order_.begin(), button_order_.end(),
256 [&](const auto& pair) {
257 return pair.first == button_name.data();
258 }),
259 button_order_.end());
260 draco_->robot_model().meshcat()->DeleteButton(button_name.data());
261 }
269 void AddButton(const std::string_view button_name, int idx = -1,
270 const std::string_view keycode = "") {
271 DRAKE_THROW_UNLESS(draco_ != nullptr);
272 auto it {button_order_.begin()};
273 if (idx < 0) {
274 // support backwards indexing
275 it = button_order_.end() + idx + 1;
276 } else {
277 it += idx;
278 }
279 if (it < button_order_.begin() || it > button_order_.end()) {
280 throw std::runtime_error(
281 fmt::format("Invalid button index: {} for button order of size {}",
282 idx, button_order_.size()));
283 }
284 button_order_.insert(it, {button_name.data(), keycode.data()});
285 ResetButtons();
286 }
