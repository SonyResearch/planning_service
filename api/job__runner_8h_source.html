67 inline bool ResultIsReady(const request_id_t& id) const {
68 std::scoped_lock<std::mutex> lock {jobs_mtx_};
69 if (!active_jobs_.count(id)) {
70 throw std::runtime_error(fmt::format("No job with ID {} found!", id));

75 inline const std::expected<ResultType, ServiceError> RetrieveResult(
76 const request_id_t& id) {
77 std::scoped_lock<std::mutex> lock {jobs_mtx_};
78 if (!active_jobs_.count(id)) {
79 throw std::runtime_error(fmt::format("No job with ID {} found!", id));
80 }
81 if (!common::utils::future_result_ready(active_jobs_.at(id))) {
82 throw std::runtime_error(fmt::format("Job {} is not ready!", id));
83 }
84 if (!active_jobs_.at(id).valid()) {
85 return std::unexpected(ServiceError(
86 ServiceErrorCode::JOB_THREAD_INVALIDATED,
87 fmt::format("JobManager:RetrieveResult: Thread at ID {} has "
88 "been invalidated!",
89 id)));
90 }
91 try {
92 const auto result_opt {active_jobs_.at(id).get()};
93 active_jobs_.erase(id);
94 return result_opt;
95 } catch (const std::exception& e) {
96 active_jobs_.erase(id);
97 return std::unexpected(ServiceError(
98 ServiceErrorCode::JOB_THREAD_EXCEPTION,
99 fmt::format("JobManager:RetrieveResult: Thread at ID {} threw an "
100 "exception: {}",
101 id, e.what())));
102 }
103 }

112 inline void WaitForResult(
113 const request_id_t& id,
114 const std::optional<uint16_t> timeout_ms_opt = std::nullopt) const {
115 std::scoped_lock<std::mutex> lock {jobs_mtx_};
116 if (!active_jobs_.count(id)) {
117 throw std::runtime_error(fmt::format("No job with ID {} found!", id));
118 }
119 if (timeout_ms_opt) {
120 active_jobs_.at(id).wait_for(chrono_ms(timeout_ms_opt.value()));
121 } else {
122 active_jobs_.at(id).wait();
123 }
124 }
132 inline void InsertNewJob(
133 const request_id_t& id,
134 const std::function<const std::expected<ResultType, ServiceError>()>
135 job) {
136 std::scoped_lock<std::mutex> lock {jobs_mtx_};
137 if (active_jobs_.count(id)) {
138 throw std::runtime_error(
139 fmt::format("Job with ID {} has already been started", id));
140 }
141 if (active_jobs_.size() >= capacity_) {
142 throw std::runtime_error("Already at maximum number of active jobs!");
143 }
144 active_jobs_.emplace(id, std::async(std::launch::async, job));
145 }

164 inline bool Contains(const request_id_t& id) const {
