85inline const proto::RigidTransform RigidTransformToProto(
86 const drake::math::RigidTransformd& X_AB) {
87 proto::RigidTransform msg;
88 const auto& translation {X_AB.translation()};
89 const auto& quat {X_AB.rotation().ToQuaternion()};
90 proto::Vector3 translation_pb;
91 translation_pb.set_x(translation(0));
92 translation_pb.set_y(translation(1));
93 translation_pb.set_z(translation(2));
94 *msg.mutable_translation() = translation_pb;
95 proto::Quaternion quat_pb;
96 quat_pb.set_w(quat.w());
97 quat_pb.set_x(quat.x());
98 quat_pb.set_y(quat.y());
99 quat_pb.set_z(quat.z());
100 *msg.mutable_quat() = quat_pb;
101 return msg;
102}

164inline const motion::ConstraintsAdapter ProtoToConstraintsAdapter(
165 const proto::ConstraintsSet& constraints_pb) {
166 motion::ConstraintsAdapter constraints_adapter;
167 // default collision checker
168 constraints_adapter.collision_checker = motion::CollisionCheckerAdapter();
169 // populate position constraints
170 if (constraints_pb.pos_constraints_size()) {
171 std::vector<motion::PositionConstraintAdapter> constraints_vec;
172 for (const auto& constraint : constraints_pb.pos_constraints()) {
173 constraints_vec.push_back(ProtoPositionConstraintAdapter(constraint));
174 }
175 constraints_adapter.position_constraints = constraints_vec;
176 }
177 // populate angle constraints
178 if (constraints_pb.angle_constraints_size()) {
179 std::vector<motion::AngleBetweenVectorsConstraintAdapter> constraints_vec;
180 for (const auto& constraint : constraints_pb.angle_constraints()) {
181 constraints_vec.push_back(
182 ProtoToAngleBetweenVectorsConstraintAdapter(constraint));
183 }
184 constraints_adapter.angle_constraints = constraints_vec;
185 }
186 return constraints_adapter;
187}

189inline const service::MeshFile ProtoToMeshFile(const proto::MeshFile& mesh_pb) {
190 std::string extension;
191 switch (mesh_pb.format()) {
192 case proto::MeshFormat::MESH_FORMAT_OBJ:
193 extension = ".obj";
194 break;
195 case proto::MeshFormat::MESH_FORMAT_MTL:
196 extension = ".mtl";
197 break;
198 case proto::MeshFormat::MESH_FORMAT_GLTF:
199 extension = ".gltf";
200 break;
201 case proto::MeshFormat::MESH_FORMAT_BIN:
202 extension = ".bin";
203 break;
204 default:
205 throw std::runtime_error(fmt::format(
206 "ProtoToMeshFile: Mesh format: {} is not currently supported!",
207 magic_enum::enum_name(mesh_pb.format())));
208 }
209 bool collision;
210 switch (mesh_pb.type()) {
211 case proto::MeshType::MESH_TYPE_COLLISION:
212 collision = true;
213 break;
214 case proto::MeshType::MESH_TYPE_VISUAL:
215 collision = false;
216 break;
217 default:
218 throw std::runtime_error(fmt::format(
219 "ProtoToMeshFile: Mesh type: {} is not currently supported!",
220 magic_enum::enum_name(mesh_pb.type())));
221 }
222 service::MeshFile mesh {.name = mesh_pb.name(),
223 .extension = extension,
224 .package_name = mesh_pb.package_name(),
225 .parent_path = mesh_pb.parent_path(),
226 .collision = collision,
227 .content = mesh_pb.content()};
228 return mesh;
229}

231inline const service::Model ProtoToModel(const proto::Model& model_pb) {
232 const auto& file_pb {model_pb.file()};
233 service::ModelFormat format;
234 switch (file_pb.format()) {
235 case proto::ModelFormat::MODEL_FORMAT_URDF:
236 format = service::ModelFormat::URDF;
237 break;
238 default:
239 throw std::runtime_error(fmt::format(
240 "ProtoToModel: Model of format: {} is not currently supported!",
241 magic_enum::enum_name(file_pb.format())));

244 .format = format,
245 .package_name = file_pb.package_name(),
246 .parent_path = file_pb.parent_path(),
247 .content = file_pb.content()};
248 std::vector<service::MeshFile> meshes;

252 service::Model model {.name = model_pb.name(),
253 .file = file,
254 .meshes = meshes,
255 .default_joint_positions = {},
256 .default_free_body_pose = {}};
257 return model;

263inline const service::PlanContext ProtoToPlanContext(
264 const proto::PlanContext& context_pb) {
265 const auto name {context_pb.name()};
266 if (name.empty()) {
267 throw std::runtime_error("No system name provided!");
268 }
269 if (const auto& context_id {context_pb.id().value()}; context_id != 0) {
270 return service::PlanContext(context_id);
271 } else {
272 std::vector<service::Model> context_models;
273 for (const auto& model : context_pb.models()) {
274 context_models.push_back(ProtoToModel(model));
275 }
276 std::string metadata_json_str;
277 const auto status {google::protobuf::util::MessageToJsonString(
278 context_pb.metadata(), &metadata_json_str)};
279 if (!status.ok()) {
280 throw std::runtime_error("Failed to convert Protobuf message to JSON: "
281 + status.ToString());
282 }
283 motion::ConstraintsAdapter constraints_adapter;
284 if (!context_pb.constraints_raw().empty()) {
285 constraints_adapter =
286 drake::yaml::LoadYamlString<motion::ConstraintsAdapter>(
287 context_pb.constraints_raw());
288 } else {
289 constraints_adapter = ProtoToConstraintsAdapter(context_pb.constraints());
290 }
291 return service::PlanContext(
292 name,
293 drake::yaml::LoadYamlString<drake::multibody::parsing::ModelDirectives>(
294 context_pb.model_directive_raw()),
295 context_models, constraints_adapter, json::parse(metadata_json_str));
296 }

339inline const service::iris::IrisBuildRequestAdapter
340ProtoToIrisBuildRequestAdapter(const proto::StartBuildRequest* req) {
341 std::vector<system_conf_t> sysconf_vec;
342 for (const auto& config_pb : req->sysconf_vec().data()) {
343 sysconf_vec.push_back(utils::ProtoToSystemConf(config_pb));
344 }
345 std::vector<system_conf_edge_t> sysconf_edge_vec;
346 for (const auto& edge_pb : req->sysconf_edge_vec().data()) {
347 sysconf_edge_vec.push_back(utils::ProtoToSystemConfEdge(edge_pb));
348 }
349
350 std::vector<draco::PlanningProblemDef> planning_problem_vec;
351 for (const auto& problem_pb : req->problem_def_vec().data()) {
352 planning_problem_vec.push_back(ProtoToPlanningProblemDef(problem_pb));
353 }
354
355 return service::iris::IrisBuildRequestAdapter(
356 req->id(), ProtoToPlanContext(req->context()), req->num_problems(),
357 ProtoToIrisBuildJobType(req->seed_data_case()), sysconf_vec,
358 sysconf_edge_vec, planning_problem_vec);
359}
