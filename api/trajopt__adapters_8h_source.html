24struct CompositeBezierTrajectoryAdapter {
25 std::vector<double> start_times_vec {};
26 std::vector<double> end_times_vec {};
27 std::vector<Eigen::MatrixXd> control_points_vec {};
28
30 template <typename Archive>
31 void Serialize(Archive* a) {
32 a->Visit(DRAKE_NVP(start_times_vec));
33 a->Visit(DRAKE_NVP(end_times_vec));
34 a->Visit(DRAKE_NVP(control_points_vec));
35 }
36 static drake::trajectories::CompositeTrajectory<double>
37 AdapterToCompositeTrajectory(
38 const CompositeBezierTrajectoryAdapter& adapter) {
39 std::vector<traj_ptr_t> subtrajs;
40 for (size_t i = 0; i < adapter.start_times_vec.size(); ++i) {
41 const auto& start_time = adapter.start_times_vec[i];
42 const auto& end_time = adapter.end_times_vec[i];
43 const auto& control_points = adapter.control_points_vec[i];
44 traj_ptr_t traj_ptr =
45 drake::copyable_unique_ptr<drake::trajectories::Trajectory<double>>(
46 new drake::trajectories::BezierCurve<double>(start_time, end_time,
47 control_points));
48 subtrajs.push_back(traj_ptr);
49 }
50 return drake::trajectories::CompositeTrajectory<double>(subtrajs);
51 }
52 static CompositeBezierTrajectoryAdapter CompositeTrajectoryToAdapter(
53 const drake::trajectories::CompositeTrajectory<double>& trajectory) {
54 CompositeBezierTrajectoryAdapter adapter;
55 for (int i = 0; i < trajectory.get_number_of_segments(); ++i) {
56 const auto& subtraj = trajectory.segment(i);
57 adapter.start_times_vec.push_back(subtraj.start_time());
58 adapter.end_times_vec.push_back(subtraj.end_time());
59 const auto& bezier_curve =
60 dynamic_cast<const drake::trajectories::BezierCurve<double>&>(
61 subtraj);
62 adapter.control_points_vec.push_back(bezier_curve.control_points());
63 }
64 return adapter;
65 }
66 static void SaveYamlFile(
67 const std::string& filename,
68 const drake::trajectories::CompositeTrajectory<double>& trajectory) {
69 drake::yaml::SaveYamlFile(filename,
70 CompositeTrajectoryToAdapter(trajectory));
71 }
72
73 static drake::trajectories::CompositeTrajectory<double> LoadYamlFile(
74 const std::string& filename) {
75 const auto adapter =
76 drake::yaml::LoadYamlFile<CompositeBezierTrajectoryAdapter>(filename);
77 return AdapterToCompositeTrajectory(adapter);
78 }
79};
