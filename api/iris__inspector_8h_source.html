274 template <typename FormatContext>
275 auto format(const motion::iris::IrisInspector::CoverageResult& result,
276 FormatContext& ctx) const {
277 int total_points = result.uncovered + result.volume_covered
278 + result.visibility_covered
279 + result.violates_constraints;
280 const double volume_coverage_percentage =
281 100.0 * (result.volume_covered)
282 / static_cast<double>(total_points - result.violates_constraints);
283 const double visibility_coverage_percentage =
284 100.0 * (result.visibility_covered)
285 / static_cast<double>(total_points - result.violates_constraints);
286 return fmt::format_to(
287 ctx.out(),
288 "Volume covered: \t\t{}/{} ({:.2f}%) \nVisibility covered: \t\t{}/{} "
289 "({:.2f}%) \nUncovered: \t\t\t {}/{} ({:.2f}%)\nViolates constraints: "
290 "\t\t {}/{} ({:.2f}%)",
291 result.volume_covered, total_points - result.violates_constraints,
292 volume_coverage_percentage, result.visibility_covered,
293 total_points - result.violates_constraints,
294 visibility_coverage_percentage, result.uncovered,
295 total_points - result.violates_constraints,
296 100.0 * result.uncovered
297 / static_cast<double>(total_points - result.violates_constraints),
298 result.violates_constraints, total_points,
299 100.0 * result.violates_constraints
300 / static_cast<double>(total_points));
301 }
