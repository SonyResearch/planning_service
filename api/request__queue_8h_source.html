72 inline void Enqueue(const RequestType& request) {
73 std::scoped_lock<std::mutex> lock {queue_mtx_};
74 if (queued_requests_.size() >= capacity_) {
75 throw queue_at_capacity_error();
76 }
77 if (queued_ids_.count(request.id)) {
78 logging::log()->debug(
79 "RequestQueue:Enqueue: Request with ID {} already in queue. Ignoring",
80 request.id);
81 }
82 // add request to the queue
83 queued_requests_.push_back(request);
84 // add ID to the set of queued IDs to track duplicate ID requests
85 queued_ids_.insert(request.id);
86 }
88 inline const RequestType& PeekNextRequest() const {
89 std::scoped_lock<std::mutex> lock {queue_mtx_};
90 if (queued_requests_.size() == 0) {
91 throw queue_empty_error();
92 }
93 return queued_requests_.front();
94 }
97 inline const RequestType PopNextRequest() {
98 std::scoped_lock<std::mutex> lock {queue_mtx_};
99 if (queued_requests_.size() == 0) {
100 throw queue_empty_error();
101 }
102 const auto req {queued_requests_.front()};
103 // destroy first element
104 queued_requests_.pop_front();
105 // remove ID from list
106 queued_ids_.erase(req.id);
107 return req;
108 }
109
111 inline bool Empty() const {
112 std::scoped_lock<std::mutex> lock {queue_mtx_};
113 return queued_requests_.empty();
114 }
116 inline size_t Size() const {
117 std::scoped_lock<std::mutex> lock {queue_mtx_};
118 return queued_requests_.size();
119 }
120
122 inline size_t Capacity() const {
123 return capacity_;
124 }
125
126 private:
127 mutable std::mutex queue_mtx_;
128 const size_t capacity_ {1};
129 // double-ended queue to store and retrieve requests
130 std::deque<RequestType> queued_requests_ {};
131 // set of ids in the queue for quick lookup
132 std::set<request_id_t> queued_ids_ {};
