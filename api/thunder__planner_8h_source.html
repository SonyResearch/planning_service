179 // create a vector of vertex indices
180 std::vector<unsigned int> vertex_indices;

182 for (unsigned int i = 0; i < planner_data_->numVertices(); i++) {

186 logging::log()->debug(
187 "ThunderPlanner:SparseAdjacencyMatrixOfRoadmap: Created adjacency "
188 "matrix of size {}x{} with {} non-zero entries",
189 matrix.rows(), matrix.cols(), matrix.nonZeros());
190 return matrix;
