130 const std::expected<IrisBuildResult, ServiceError> BuildRegions(
131 const draco::PlanContextId& context_id,
132 const IrisBuildRequestAdapter& req) {
133 try {
134 switch (req.job_type) {
135 case IrisBuildJobType::IRIS_FROM_EDGES: {
136 return BuildRegionsFromEdges(context_id, req.sysconf_edge_vec);
137 }
138 case IrisBuildJobType::IRIS_FROM_CONFIGS: {
139 return BuildRegionsFromSystemConfs(context_id, req.sysconf_vec);
140 break;
141 }
142 case IrisBuildJobType::IRIS_FROM_ROADMAP: {
143 return BuildRegionsFromRoadmap(context_id, req.num_problems);
144 }
145 default:
146 return std::unexpected(ServiceError(
147 ServiceErrorCode::JOB_TYPE_UNSUPPORTED,
148 fmt::format("Job type {} is not currently supported!",
149 magic_enum::enum_name(req.job_type))));
150 }
151 } catch (const std::exception& e) {
152 return std::unexpected(ServiceError(
153 ServiceErrorCode::JOB_THREAD_EXCEPTION,
154 fmt::format("Job failed due to exception: {}", e.what())));
155 }
156 }
