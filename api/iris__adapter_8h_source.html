214 template <typename Archive>
215 void Serialize(Archive* a) {
216 a->Visit(DRAKE_NVP(regions_vec_));
217 a->Visit(DRAKE_NVP(intersections_vec_));
218
219 if (!intersections_vec_.has_value() && !regions_vec_.empty()) {
220 const auto regions_vec_copy = regions_vec_;
221 *this = IrisRegionsAdapter(); // Reset the adapter to an empty state
222 // Re-add the regions to the empty adapter.
223 // Need to populate intersections_vec_ if it is not set.
224 logging::log()->warn(
225 "IrisRegionsAdapter::Serialize: intersections_vec_ is not set, "
226 "populating it with no continuous revolute joints and 1 "
227 "intersection samples.");
228 // Need to add intersections to "this" object.
229 for (int i = 0; i < std::ssize(regions_vec_copy); ++i) {
230 const auto& region = regions_vec_copy.at(i);
231 this->AddRegion(region.set(), region.name(), region.constraints_hash());
232 }
233 DRAKE_DEMAND(intersections_vec_.has_value());
234 logging::log()->info(
235 "IrisRegionsAdapter::Serialize: populated intersections_vec_ with "
236 "empty intersections. Now it has {} regions and {} intersections.",
237 regions_vec_.size(), intersections_vec_.value().size());
238 }
239 }
