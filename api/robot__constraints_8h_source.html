466 template <typename Scalar>
467 void SetCollisionMatrix(Eigen::Matrix<Scalar, -1, -1>& m, Scalar val,
468 std::string_view body_or_group_1,
469 std::string_view body_or_group_2,
470 const std::string caller = "") const {
471 const auto is_group_1 {
472 robot_model_.HasFilterGroupName(body_or_group_1.data())};
473 const auto is_body_1 {
474 robot_model_.HasScopedFrameNamed(body_or_group_1.data())};
475 const auto is_group_2 {
476 robot_model_.HasFilterGroupName(body_or_group_2.data())};
477 const auto is_body_2 {
478 robot_model_.HasScopedFrameNamed(body_or_group_2.data())};
479 auto msg_prefix {caller.empty() ? "SetCollisionMatrix" : caller};
480 if (!body_or_group_1.empty() && !(is_group_1 || is_body_1)) {
481 throw std::runtime_error(fmt::format("{}: body or group 1 '{}' not found",
482 msg_prefix, body_or_group_1));
483 }
484 if (!body_or_group_2.empty() && !(is_group_2 || is_body_2)) {
485 throw std::runtime_error(fmt::format("{}: body or group 2 '{}' not found",
486 msg_prefix, body_or_group_2));
487 }
488 // If both empty, warn that this will set the entire matrix to the value.
489 if (body_or_group_1.empty() && body_or_group_2.empty()) {
490 logging::log()->warn(
491 "{}: Both elements are empty. Setting entire matrix to value {:.2f}. "
492 "This may have unintended consequences.",
493 msg_prefix, static_cast<double>(val));
494 }
495 logging::log()->debug(
496 "RobotConstraints:{}: Setting value for {} '{}', {} '{}' to {:.2f}",
497 msg_prefix, is_body_1 ? "body" : "group",
498 body_or_group_1.empty() ? "ALL" : body_or_group_1,
499 is_body_2 ? "body" : "group",
500 body_or_group_2.empty() ? "ALL" : body_or_group_2,
501 static_cast<double>(val));
502 auto indices_1 {robot_model_.GetBodyIndices()};
503 if (is_group_1) {
504 indices_1 = robot_model_.GetBodyIndicesFromFilterGroupName(
505 body_or_group_1.data());
506 } else if (is_body_1) {
507 indices_1 = {robot_model_.GetScopedFrameByName(body_or_group_1.data())
508 .body()
509 .index()};
510 }
511 // If no group/body B is provided, filter group/body A against
512 // all other bodies in the model.
513 auto indices_2 {robot_model_.GetBodyIndices()};
514 if (is_group_2) {
515 indices_2 = robot_model_.GetBodyIndicesFromFilterGroupName(
516 body_or_group_2.data());
517 } else if (is_body_2) {
518 indices_2 = {robot_model_.GetScopedFrameByName(body_or_group_2.data())
519 .body()
520 .index()};
521 }
522 for (const auto& idx_1 : indices_1) {
523 for (const auto& idx_2 : indices_2) {
524 const auto i {static_cast<int>(idx_1)};
525 const auto j {static_cast<int>(idx_2)};
526 if (i == j) {
527 continue;
528 }
529 m(i, j) = val;
530 m(j, i) = val;
531 }
532 }
533 }
