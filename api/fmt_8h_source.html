43template <typename T, typename U>
44struct formatter<std::map<T, U>> : formatter<std::string_view> {
45 template <typename FormatContext>
46 auto format(const std::map<T, U>& m, FormatContext& ctx) const {
47 if (m.empty()) {
48 return formatter<std::string_view>::format("{{ }}", ctx);
49 }
50 std::vector<std::string> v;
51 v.reserve(m.size());
52 for (const auto& [k, val] : m) {
53 v.push_back(fmt::format("{}: {}", k, val));
54 }
55 return formatter<std::string_view>::format(
56 fmt::format("{{ {} }}", fmt::join(v, ", ")), ctx);
57 }
58};
