79 std::expected<bool, ServiceError> QueueRequest(const RequestType& request) {

169 void Run() {
170 logging::log()->info("ResourceManager:Run: Polling queue for new requests");
171 while (!stop_triggered_) {
172 Cleanup();
173 if (request_queue_->Empty()
174 || job_runner_->Size() == job_runner_->Capacity()) {
175 continue;
176 }
177 auto result {RunOnce()};
178 if (!result) {
179 logging::log()->error(
180 "ResourceManager:Run: Internal error attempting to start "
181 "next job: {}",
182 result.error());
183 }
184 std::this_thread::sleep_for(
185 chrono_ms(static_cast<uint16_t>(1e3 / poll_freq_hz_)));
186 }
187 logging::log()->info("ResourceManager:Run: Stopping resource manager");
188 }
