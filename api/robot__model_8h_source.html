49struct RobotMeshcatParams {
50 // Ports which may not be used, typically because they are
51 // allocated to another process.
52 std::optional<std::vector<int>> reserved_ports {};
53 // Optionally specify an explicit port to use.
54 std::optional<int> port {};
55 // @warning: if set to true, the meshcat visualizer can be very slow
56 // on some machines. It is recommended to set this to false on a
57 // laptop that has a weak Graphics card.
58 bool visual {false};
59 // load collision geos
60 bool collision {false};
61 // colors of the bodies
62 body_name_color_map_t color_map {};
63 // colors of Cartesian-projected PRMs; if a color is not defined here, then a
64 // random one will be selected instead
65 body_name_color_map_t prm_color_map {};
66 // the list of frames to be displayed in meshcat
67 std::vector<std::string> end_effector_frame_vec {};
68 // Load them from YAML file
69 template <typename Archive>
70 void Serialize(Archive* a) {
71 a->Visit(DRAKE_NVP(reserved_ports));
72 a->Visit(DRAKE_NVP(port));
73 a->Visit(DRAKE_NVP(visual));
74 a->Visit(DRAKE_NVP(collision));
75 a->Visit(DRAKE_NVP(color_map));
76 a->Visit(DRAKE_NVP(prm_color_map));
77 a->Visit(DRAKE_NVP(end_effector_frame_vec));
78 }
79};

657 std::string fmt_msg {};
